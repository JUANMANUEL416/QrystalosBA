5.2.1 Fragmentos SQL (especificación) — no ejecutar desde BA.

-- SPQ_DISTRIHPRED_SOAT · MOVER_ITEM_PARCIAL
-- AFTER leer @NOPRESTACION / @NOITEM y antes de mover:

IF EXISTS (
  SELECT 1
  FROM HPRED D
  INNER JOIN HPRE H ON H.NOPRESTACION = D.NOPRESTACION
  WHERE D.NOPRESTACION = @NOPRESTACION AND D.NOITEM = @NOITEM
    AND (COALESCE(H.PEDIDOINV,0) = 1 OR COALESCE(D.PEDIDOINV,0) = 1)
)
BEGIN
  SELECT 'KO' OK
  SELECT ERROR = N'Prestación con pedido a inventario: primero deshaga el pedido'
  RETURN
END

-- SPQ_DISTRIHPRED_SOAT · MOVER_MASIVO
-- BEFORE cursor/UPDATE masivo (respetar @GFACSOAT_ORIGEN y filtros fecha/prefijo/usuario):

IF EXISTS (
  SELECT 1
  FROM HPRED D
  INNER JOIN HPRE H ON H.NOPRESTACION = D.NOPRESTACION
  WHERE H.NOADMISION = @NOADMISION
    AND COALESCE(D.FACTURADA,0) = 0
    AND COALESCE(D.NOCOBRABLE,0) = 0
    AND D.GFACSOAT = @GFACSOAT_ORIGEN
    -- + predicados de fecha/prefijo/usuario del METODO
    AND (COALESCE(H.PEDIDOINV,0) = 1 OR COALESCE(D.PEDIDOINV,0) = 1)
)
BEGIN
  SELECT 'KO' OK
  SELECT ERROR = N'Prestación con pedido a inventario: primero deshaga el pedido'
  RETURN
END

-- Validación soporte / preversión:
SELECT H.NOPRESTACION, H.PREFIJO, H.PEDIDOINV, H.ESDEINV, COUNT(*) AS items
FROM HPRE H
INNER JOIN HPRED D ON D.NOPRESTACION = H.NOPRESTACION
WHERE H.NOADMISION = @NOADMISION AND COALESCE(H.PEDIDOINV,0) = 1
GROUP BY H.NOPRESTACION, H.PREFIJO, H.PEDIDOINV, H.ESDEINV;
